add_library(waddamburo_media SHARED src/media.c src/video.c src/transcode.c)
add_library(Waddamburo::Media ALIAS waddamburo_media)

waddamburo_configure_native_target(waddamburo_media)
if(MINGW)
  set_target_properties(waddamburo_media PROPERTIES PREFIX "")
endif()
target_compile_definitions(waddamburo_media PRIVATE WADDAMBURO_MEDIA_BUILDING_LIBRARY=1)
target_include_directories(waddamburo_media
  PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>)

option(WADDAMBURO_MEDIA_FFMPEG "Build the media decoder with FFmpeg" OFF)
set(WADDAMBURO_FFMPEG_ROOT "" CACHE PATH "Optional FFmpeg installation prefix")
if(WADDAMBURO_MEDIA_FFMPEG)
  if(WADDAMBURO_PACKAGE)
    set(_ffmpeg_prefix "${CMAKE_BINARY_DIR}/ffmpeg/prefix")
    target_include_directories(waddamburo_media SYSTEM PRIVATE "${_ffmpeg_prefix}/include")
    foreach(_library avformat avcodec swresample swscale avutil)
      target_link_libraries(waddamburo_media PRIVATE "${_ffmpeg_prefix}/lib/lib${_library}.a")
    endforeach()
    # avcodec's libopus encoder (the decoded-audio cache), static in the same prefix.
    target_link_libraries(waddamburo_media PRIVATE
      "${_ffmpeg_prefix}/lib/${CMAKE_STATIC_LIBRARY_PREFIX}opus${CMAKE_STATIC_LIBRARY_SUFFIX}")
    add_dependencies(waddamburo_media waddamburo_ffmpeg)
  elseif(WADDAMBURO_FFMPEG_ROOT)
    set(_ffmpeg_include "${WADDAMBURO_FFMPEG_ROOT}/include")
    if(NOT EXISTS "${_ffmpeg_include}/libavcodec/avcodec.h")
      message(FATAL_ERROR "WADDAMBURO_FFMPEG_ROOT does not contain FFmpeg headers")
    endif()
    set(_ffmpeg_library_paths "${WADDAMBURO_FFMPEG_ROOT}/lib")
    if(WIN32)
      list(APPEND _ffmpeg_library_paths "${WADDAMBURO_FFMPEG_ROOT}/bin")
    endif()
    find_library(_avformat avformat PATHS ${_ffmpeg_library_paths} NO_DEFAULT_PATH REQUIRED)
    find_library(_avcodec avcodec PATHS ${_ffmpeg_library_paths} NO_DEFAULT_PATH REQUIRED)
    find_library(_swresample swresample PATHS ${_ffmpeg_library_paths} NO_DEFAULT_PATH REQUIRED)
    find_library(_avutil avutil PATHS ${_ffmpeg_library_paths} NO_DEFAULT_PATH REQUIRED)
    find_library(_swscale swscale PATHS ${_ffmpeg_library_paths} NO_DEFAULT_PATH REQUIRED)
    target_include_directories(waddamburo_media SYSTEM PRIVATE "${_ffmpeg_include}")
    target_link_libraries(waddamburo_media PRIVATE
      "${_avformat}" "${_avcodec}" "${_swresample}" "${_swscale}" "${_avutil}")
  else()
    find_package(PkgConfig REQUIRED)
    pkg_check_modules(WADDAMBURO_FFMPEG REQUIRED IMPORTED_TARGET
      libavformat libavcodec libswresample libswscale libavutil)
    target_link_libraries(waddamburo_media PRIVATE PkgConfig::WADDAMBURO_FFMPEG)
  endif()
  target_compile_definitions(waddamburo_media PRIVATE WADDAMBURO_MEDIA_HAS_FFMPEG=1)
endif()

if(WADDAMBURO_MEDIA_VGMSTREAM)
  if(NOT WADDAMBURO_MEDIA_FFMPEG)
    message(FATAL_ERROR "The vgmstream backend currently requires the FFmpeg backend for output conversion")
  endif()
  target_include_directories(waddamburo_media SYSTEM PRIVATE "${WADDAMBURO_VGMSTREAM_INCLUDE_DIR}")
  # vgmstream first: static archives resolve left to right.
  target_link_libraries(waddamburo_media PRIVATE "${WADDAMBURO_VGMSTREAM_LIBRARY}")
  add_dependencies(waddamburo_media waddamburo_vgmstream_build)
  target_compile_definitions(waddamburo_media PRIVATE WADDAMBURO_MEDIA_HAS_VGMSTREAM=1)
endif()
if(WADDAMBURO_PACKAGE)
  if(MINGW)
    target_link_libraries(waddamburo_media PRIVATE bcrypt)
    target_link_options(waddamburo_media PRIVATE -static)
  elseif(UNIX)
    find_package(Threads REQUIRED)
    target_link_libraries(waddamburo_media PRIVATE m Threads::Threads)
    target_link_options(waddamburo_media PRIVATE -Wl,--exclude-libs,ALL)
  endif()
endif()
